Modals

Modal dialogs with backdrop and scroll lock.

Basic Modal

Modal Title

This is a basic modal dialog. Click the backdrop or the close button to dismiss.

Code Example

<!-- Modal Structure -->
<div class="ax-modal" data-open="false">
  <div class="ax-modal__backdrop"></div>
  <div class="ax-modal__dialog">
    <div class="ax-modal__header">
      <h2 class="ax-modal__title">Title</h2>
      <button class="ax-modal__close">×</button>
    </div>
    <div class="ax-modal__body">
      <p>Content here</p>
    </div>
    <div class="ax-modal__footer">
      <button class="ax-btn ax-btn--secondary">Cancel</button>
      <button class="ax-btn ax-btn--primary">Confirm</button>
    </div>
  </div>
</div>

<!-- Toggle with data-open attribute -->
<button onclick="modal.setAttribute('data-open', 'true')">
  Open Modal
</button>
